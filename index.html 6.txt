    function selectOption(option) {
      document.getElementById("home").style.display = "none";
      
      if (option === 'diagnosis') {
        showDiagnosisCategories();
        document.getElementById("diagnosisPanel").classList.add("active");
      } else if (option === 'pricing') {
        document.getElementById("pricingPanel").classList.add("active");
      } else if (option === 'technician') {
        document.getElementById("technicianPanel").classList.add("active");
      }
    }

    function showDiagnosisCategories() {
      document.getElementById("diagnosisDetail").style.display = "none";
      const grid = document.getElementById("diagnosisGrid");
      document.getElementById("diagnosisTitle").textContent = "Arıza Kategorisi";
      
      grid.innerHTML = Object.entries(diagnosticData).map(([key, data]) => `
        <button class="card" onclick="selectDiagnosis('${key}')">
          <span class="card-icon">${data.icon}</span>
          <div class="card-title">${data.title}</div>
        </button>
      `).join("");
    }

    function selectDiagnosis(type) {
      const data = diagnosticData[type];
      document.getElementById("diagnosisGrid").style.display = "none";
      document.getElementById("diagnosisDetail").style.display = "block";

      document.getElementById("detailTitle").textContent = data.title;
      
      const questionsDiv = document.getElementById("detailQuestions");
      questionsDiv.innerHTML = data.questions.map((q, i) => `
        <div class="question">${i + 1}. ${q}</div>
      `).join("");

      document.getElementById("detailResult").textContent = data.result;

      const message = `Merhaba, TEHAI üzerinden arıza bildirimi oluşturuyorum.%0A%0AKategori: ${data.title}%0A%0AKontrol soruları:%0A${data.questions.map((q, i) => `${i + 1}. ${q}`).join("%0A")}%0A%0A${data.result}`;
      document.getElementById("detailWhatsapp").href = `https://wa.me/905367807059?text=${message}`;

      window.scrollTo(0, 0);
    }

    function goBack() {
      document.getElementById("home").style.display = "block";
      document.getElementById("diagnosisPanel").classList.remove("active");
      document.getElementById("pricingPanel").classList.remove("active");
      document.getElementById("technicianPanel").classList.remove("active");
      document.getElementById("diagnosisGrid").style.display = "grid";
      document.getElementById("diagnosisDetail").style.display = "none";
      window.scrollTo(0, 0);
    }
