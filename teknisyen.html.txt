  <script>
    let state = {
      currentStep: 1,
      phone: null,
      token: null,
      category: null,
      technician: null,
      latitude: null,
      longitude: null,
      address: null
    };

    const categories = {
      elektrik: { icon: '⚡', title: 'Elektrik Arızası' },
      kamera: { icon: '📷', title: 'Kamera Sistemi' },
      internet: { icon: '📡', title: 'İnternet / Ağ' },
      uydu: { icon: '📺', title: 'Uydu / TV' },
      motor: { icon: '🚪', title: 'Kapı / Motor' }
    };

    // Init
    window.addEventListener('DOMContentLoaded', () => {
      renderCategories();
      getLocation();
    });

    // OTP Gönder
    async function sendOTP() {
      const phone = document.getElementById('phoneInput').value.replace(/\D/g, '');
      if (phone.length !== 10) {
        alert('Geçerli bir telefon numarası girin');
        return;
      }

      state.phone = '+90' + phone;

      try {
        const res = await fetch('/api/auth', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ action: 'send-otp', phone: state.phone })
        });

        const data = await res.json();
        if (data.success) {
          goToStep(1.5);
          // Dev modunda kodu göster
          if (data.code) {
            document.getElementById('otpMessage').textContent = `📌 Test kodu: ${data.code}`;
          } else {
            document.getElementById('otpMessage').textContent = 'Kodu SMS ile aldınız';
          }
        } else {
          alert(data.message || 'Hata oluştu');
        }
      } catch (error) {
        console.error('Error:', error);
        alert('Bağlantı hatası');
      }
    }

    // OTP Doğrula
    async function verifyOTP() {
      const code = document.getElementById('otpInput').value;
      if (code.length !== 6) {
        alert('6 haneli kodu girin');
        return;
      }

      try {
        const res = await fetch('/api/auth', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ action: 'verify-otp', phone: state.phone, code })
        });

        const data = await res.json();
        if (data.success) {
          state.token = data.token;
          goToStep(2);
        } else {
          alert(data.message || 'Hata oluştu');
        }
      } catch (error) {
        console.error('Error:', error);
        alert('Bağlantı hatası');
      }
    }

    // Kategorileri Render Et
    function renderCategories() {
      const grid = document.getElementById('categoriesGrid');
      grid.innerHTML = Object.entries(categories).map(([key, cat]) => `
        <button class="card" onclick="selectCategory('${key}')">
          <div class="card-icon">${cat.icon}</div>
          <div class="card-title">${cat.title}</div>
        </button>
      `).join('');
    }

    // Kategori Seç
    function selectCategory(key) {
      state.category = key;
      document.querySelectorAll('#step2 .card').forEach(c => c.classList.remove('selected'));
      event.target.closest('.card').classList.add('selected');
      document.getElementById('continueBtn').disabled = false;
    }

    // Konum Al
    function getLocation() {
      if ('geolocation' in navigator) {
        navigator.geolocation.getCurrentPosition(
          (position) => {
            state.latitude = position.coords.latitude;
            state.longitude = position.coords.longitude;
            console.log('Konum:', state.latitude, state.longitude);
          },
          (error) => {
            console.error('Konum hatası:', error);
            // Mock konum (İstanbul)
            state.latitude = 41.0082;
            state.longitude = 28.9784;
          }
        );
      }
    }

    // Step Geçişi
    function goToStep(step) {
      document.querySelectorAll('.step').forEach(s => s.classList.remove('active'));
      
      if (step === 1.5) {
        document.getElementById('step1b').classList.add('active');
      } else {
        document.getElementById(`step${Math.ceil(step)}`).classList.add('active');
      }

      state.currentStep = step;
      window.scrollTo(0, 0);

      if (step === 3) {
        loadNearbyTechnicians();
      }
    }

    function previousStep() {
      const steps = [1, 1.5, 2, 3, 4, 5];
      const currentIndex = steps.indexOf(state.currentStep);
      if (currentIndex > 0) {
        goToStep(steps[currentIndex - 1]);
      } else {
        goBack();
      }
    }

    function goBack() {
      window.location.href = '/';
    }

    function selectTechnician(id, name) {
      state.technician = { id, name };

    }

    // Servis İsteğini Gönder
    async function submitRequest() {
      const notes = document.getElementById('notesInput').value;

      const payload = {
        phone: state.phone,
        category: state.category,
        technician_id: state.technician.id,
        latitude: state.latitude,
        longitude: state.longitude,
        notes: notes,
        token: state.token
      };

      console.log('Servis isteği gönderiliyor:', payload);

      // Başarı göster
      document.getElementById('successDetails').innerHTML = `
        <div style="padding: 16px; background: rgba(255,255,255,0.03); border-radius: 12px; margin-bottom: 20px;">
          <div style="margin-bottom: 12px;"><strong>Seçilen Teknisyen:</strong> ${state.technician.name}</div>
          <div style="margin-bottom: 12px;"><strong>Kategori:</strong> ${categories[state.category].title}</div>
          <div><strong>İstek Numarası:</strong> #TEH${Date.now().toString().slice(-6)}</div>
        </div>
      `;

      goToStep(5);
    }
