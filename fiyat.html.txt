  <script>
    const kategoriler = {
      "ip-kamera": { icon: "📷", title: "IP Kamera", desc: "Standart, PoE destekli" },
      "solar-kamera": { icon: "☀️", title: "Solar Kamera", desc: "Güneş enerjili, 4G" },
      "analog-kamera": { icon: "📼", title: "Analog Kamera", desc: "AHD/HDCVI" },
      "nvr": { icon: "🖥️", title: "NVR", desc: "IP Kayıt Cihazı" },
      "dvr": { icon: "📀", title: "DVR", desc: "Analog Kayıt Cihazı" },
      "poe-switch": { icon: "🔌", title: "PoE Switch", desc: "Ağ Anahtarı" },
      "komple-set": { icon: "📦", title: "Komple Set", desc: "Sistem paketi" }
    };

    const markalarByKategori = {
      "ip-kamera": ["Hikvision", "Dahua", "HiLook", "Uniview", "EZVIZ", "Imou", "TP-Link"],
      "solar-kamera": ["Hikvision", "Dahua", "EZVIZ", "Imou"],
      "analog-kamera": ["Hikvision", "Dahua", "HiLook", "Uniview"],
      "nvr": ["Hikvision", "Dahua", "HiLook", "Uniview"],
      "dvr": ["Hikvision", "Dahua", "HiLook"],
      "poe-switch": ["BDCOM", "TP-Link", "Hikvision", "Netgear"],
      "komple-set": ["Hikvision", "Dahua", "HiLook"]
    };

    let state = {
      kategori: null,
      marka: null,
      adet: 1,
      alan: 0,
      currentStep: 1
    };

    const loadingMessages = [
      "Hikvision yetkili bayi fiyatları taranıyor…",
      "Güncel piyasa fiyatları karşılaştırılıyor…",
      "İşçilik maliyeti hesaplanıyor…",
      "Tahmininiz hazırlanıyor…"
    ];

    let messageIndex = 0;

    // Init
    window.addEventListener("DOMContentLoaded", () => {
      renderKategoriler();
      updateProgress();
    });

    // Kategorileri render et
    function renderKategoriler() {
      const container = document.getElementById("kategoriler");
      container.innerHTML = Object.entries(kategoriler).map(([key, val]) => `
        <div class="card" onclick="selectKategori('${key}')">
          <span class="card-icon">${val.icon}</span>
          <div class="card-title">${val.title}</div>
          <div class="card-desc">${val.desc}</div>
        </div>
      `).join("");
    }

    // Kategori seç
    function selectKategori(key) {
      state.kategori = key;
      document.querySelectorAll("#step1 .card").forEach(c => c.classList.remove("selected"));
      event.target.closest(".card").classList.add("selected");
      
      setTimeout(() => goToStep(2), 300);
    }

    // Markalar render
    function renderMarkalar() {
      const container = document.getElementById("markalar");
      const markalar = markalarByKategori[state.kategori] || [];
      container.innerHTML = markalar.map(m => `
        <div class="card" onclick="selectMarka('${m}')">
          <div class="card-title">${m}</div>
        </div>
      `).join("");
    }

    function selectMarka(marka) {
      state.marka = marka;
      document.querySelectorAll("#step2 .card").forEach(c => c.classList.remove("selected"));
      event.target.closest(".card").classList.add("selected");
      
      setTimeout(() => goToStep(3), 300);
    }

    // Slider
    document.addEventListener("input", (e) => {
      if (e.target.id === "adetSlider") {
        state.adet = e.target.value;
        document.getElementById("adetDisplay").textContent = state.adet;
      }
    });

    function selectAlan(value) {
      state.alan = value;
      document.querySelectorAll("#step3 .option-btn").forEach(b => b.classList.remove("selected"));
      event.target.classList.add("selected");
    }

    // Step navigasyonu
    function goToStep(step) {
      if (step === 2) renderMarkalar();
      if (step === 3) renderAlanOptions();
      if (step === 4) fetchFiyat();

      document.querySelectorAll(".step").forEach(s => s.classList.remove("active"));
      document.getElementById(`step${step}`).classList.add("active");
      state.currentStep = step;
      updateProgress();
      window.scrollTo(0, 0);
    }

    function previousStep() {
      goToStep(Math.max(1, state.currentStep - 1));
    }

    function updateProgress() {
      const percent = (state.currentStep / 5) * 100;
      document.getElementById("progressBar").style.width = percent + "%";
    }

    function renderAlanOptions() {
      const options = ["20m", "50m", "100m", "200m"];
      const container = document.getElementById("alanOptions");
      container.innerHTML = options.map(opt => `
        <button class="option-btn" onclick="selectAlan(${opt.replace('m', '')})">
          ${opt} kablo
        </button>
      `).join("");
    }

    // API çağrısı
    async function fetchFiyat() {
      messageIndex = 0;
      updateLoadingMessage();

      const messageInterval = setInterval(() => {
        messageIndex = (messageIndex + 1) % loadingMessages.length;
        updateLoadingMessage();
      }, 2000);

      try {
        const url = `/api/fiyat-tahmin?kategori=${state.kategori}&marka=${state.marka}&adet=${state.adet}&alan=${state.alan}`;
        const response = await fetch(url);
        const data = await response.json();

        clearInterval(messageInterval);

        if (data.error) {
          console.error("Hata:", data.error);
          showError();
        } else {
          displayResult(data);
          goToStep(5);
        }
      } catch (error) {
        console.error("Error:", error);
        clearInterval(messageInterval);
        showError();
      }
    }

    function updateLoadingMessage() {
      document.getElementById("loadingText").textContent = loadingMessages[messageIndex];
    }

    function displayResult(data) {
      const formatPrice = (n) => new Intl.NumberFormat("tr-TR").format(Math.round(n));

      document.getElementById("urunFiyat").textContent = 
        formatPrice((data.urun_fiyat_min + data.urun_fiyat_max) / 2) + " TL";
      document.getElementById("urunRange").textContent = 
        `${formatPrice(data.urun_fiyat_min)} - ${formatPrice(data.urun_fiyat_max)} TL`;

      document.getElementById("iscilikFiyat").textContent = 
        formatPrice((data.iscilik_min + data.iscilik_max) / 2) + " TL";
      document.getElementById("iscilikRange").textContent = 
        `${formatPrice(data.iscilik_min)} - ${formatPrice(data.iscilik_max)} TL`;

      document.getElementById("toplamFiyat").textContent = 
        `${formatPrice(data.toplam_min)} - ${formatPrice(data.toplam_max)} TL`;

      document.getElementById("modelOnerisi").textContent = data.model_onerisi || "-";
      document.getElementById("notlar").innerHTML = `
        <strong>📍 Kaynak:</strong> ${data.kaynak}<br>
        <strong>⚠️ Not:</strong> ${data.notlar}
      `;

      const msg = `Merhaba TEHAI! ${state.kategori} | ${state.marka} | ${state.adet} adet için fiyat tahmini: ${formatPrice(data.toplam_min)}-${formatPrice(data.toplam_max)} TL. Keşif randevusu istiyorum.`;
      document.getElementById("whatsappBtn").href = 
        `https://wa.me/905367807059?text=${encodeURIComponent(msg)}`;
    }

    function showError() {
      document.getElementById("urunFiyat").textContent = "Hata oluştu";
      document.getElementById("toplamFiyat").textContent = "Tekrar deneyin";
      document.getElementById("notlar").innerHTML = 
        "⚠️ Şu anda veri alınamıyor. Lütfen tekrar deneyin veya doğrudan WhatsApp'tan iletişime geçin.";
      goToStep(5);
    }
  </script>
