  <script>
    let token = localStorage.getItem('tehai_token');

    if (!token) {
      window.location.href = '/';
      throw new Error('Token yok');
    }

    window.addEventListener('DOMContentLoaded', () => {
      verifyAndLoadData();
    });

    async function verifyAndLoadData() {
      try {
        const verifyRes = await fetch('/api/auth', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ action: 'verify-token', token })
        });

        const verifyData = await verifyRes.json();
        if (!verifyData.valid) {
          localStorage.removeItem('tehai_token');
          window.location.href = '/';
          return;
        }

        document.getElementById('profilePhone').textContent = verifyData.user.phone_number;
        document.getElementById('profileRole').textContent = verifyData.user.role;

        loadRequests();
      } catch (error) {
        console.error('Error:', error);
      }
    }

    async function loadRequests() {
      try {
        const res = await fetch('/api/user/requests?token=' + token);
        const data = await res.json();

        if (data.success && data.requests.length > 0) {
          const html = data.requests.map(req => `
            <div class="request-card">
              <div class="request-header">
                <div class="request-category">${req.category.toUpperCase()}</div>
                <div class="request-status">${req.status}</div>
              </div>
              <div class="request-details">
                <div>📝 ${req.description || 'Açıklama yok'}</div>
                <div>💰 ${req.estimated_price_min || '-'} - ${req.estimated_price_max || '-'} ₺</div>
                <div>📅 ${new Date(req.created_at).toLocaleDateString('tr-TR')}</div>
              </div>
            </div>
          `).join('');
          document.getElementById('requestsList').innerHTML = html;
        } else {
          document.getElementById('requestsList').innerHTML = '<div class="empty">Henüz talep oluşturmadınız</div>';
        }
      } catch (error) {
        document.getElementById('requestsList').innerHTML = '<div class="error">Talepler alınamadı</div>';
      }
    }

    function switchTab(tab) {
      document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
      document.querySelectorAll('.content').forEach(c => c.classList.remove('active'));
      
      event.target.classList.add('active');
      document.getElementById(tab).classList.add('active');
    }

    function logout() {
      localStorage.removeItem('tehai_token');
      window.location.href = '/';
    }
  </script>
